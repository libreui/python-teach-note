# 输入三条边长，自动判断是否能构成三角形
def is_triangle(a, b, c):
    # 找到最长边
    max_side = max(a, b, c)
    sum_other = a + b + c - max_side
    if sum_other > max_side and a>0 and b>0 and c>0:
        return True
    else:
        return False

# 测试校内例题
print(is_triangle(3,4,5))  # 合法
print(is_triangle(1,2,4))  # 不合法

```

def is_right_triangle(a,b,c):
    sides = sorted([a,b,c])

def cal_tri_angle(a1,a2):
    if a1+a2 >= 180 or a1<0 or a2<0:
        return "角度不合法"
    a3 = 180 - a1 - a2
    outer_angle = a1 + a2

def perfect_square(a,b):
    res_add = (a+b)**2
    res_sub = (a-b)**2

def find_pythagorean(limit):
    res = []
    for a in range(1,limit+1):
        for b in range(a,limit+1):
            c = math.sqrt(a**2 + b**2)
            if c.is_integer():

# 1. 一次函数求值+增减性判断
def linear_func(x,k,b):
    y = k * x + b
    trend = "单调递增" if k>0 else "单调递减" if k<0 else "常函数"
    return y,trend

# 2. 求两个一次函数交点
def get_intersection(k1,b1,k2,b2):
    if k1 == k2:
        return "两直线平行，无交点/重合"
    x = (b2 - b1) / (k1 - k2)
    y = k1 * x + b1
    return round(x,2),round(y,2)

# 测试中考题型
print(linear_func(5,2,3)) # y=2x+3 x=5
print(get_intersection(2,1,-1,4)) # y=2x+1 与 y=-x+4 交点
```
